const text = "Wonderful, Joyful, Happiness, Time, Task, Apple";
const words = text.split(", "); //  розбила оригінальний рядок тексту на масив слів із використанням «,» (кома та пробіл) як роздільник. 
const matches = []; //Порожній масив під назвою `matches` створюєм для зберігання слів, які відповідають шаблону 
  
const pattern = /^(?![Aa].*).{6,}$/; 
//`(?![Aa].*)` - Негативний lookahead, який перевіряє, чи рядок не починається з «А» або «а».
//`.{6,}` - Знаходить будь-який символ (.), який повторюється 6 або більше разів ({6,}).
//`$` - Кінець рядка.


  
words.forEach(word => {
  if (pattern.test(word)) { // Метод `test()` регулярного виразу перевіряє, чи відповідає шаблон  заданому рядку 
    matches.push(word); 
  }
});
  
console.log(matches);
